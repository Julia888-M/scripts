SELECT
    {
	[Measures].[Forecast Quantity BUoM],
	[Measures].[Actual Quantity BUoM],
	[Measures].[Execution Forecast Quantity BUoM, %],
	[Measures].[Error Forecast Quantity BUoM]
	} ON COLUMNS,
    FILTER(
        NONEMPTY(
            [Calendar].[Year Month].[Year Month] * 
            [Calendar].[Year].[Year] * 
            [Demand Scenario].[Demand Scenario].[Demand Scenario] * 
            [Fact Type].[Fact Type].[Fact Type] * 
            [Article].[Article Sku].[Article Sku] * 
            [Partner].[Partner Planning Code].[Partner Planning Code],
			{
			[Measures].[Forecast Quantity BUoM],
			[Measures].[Actual Quantity BUoM],
			[Measures].[Execution Forecast Quantity BUoM, %],
			[Measures].[Error Forecast Quantity BUoM]
			}
	    ),
        [Calendar].[Year].CurrentMember.Name >= "2025"
        AND [Demand Scenario].[Demand Scenario].CurrentMember.Name = "IBP Таргет М-1"
    ) ON ROWS
FROM [Demand]
