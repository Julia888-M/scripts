SELECT 
NON EMPTY
 (
	{
	[Measures].[Forecast Net Value],
    [Measures].[Forecast Quantity BUoM]
	} 
) ON COLUMNS,
	
NON EMPTY
	( 
		{
		[Demand Scenario].[Demand Scenario].&[27],
		[Demand Scenario].[Demand Scenario].&[26],
		[Demand Scenario].[Demand Scenario].&[18],
		[Demand Scenario].[Demand Scenario].&[21],
		[Demand Scenario].[Demand Scenario].&[15]
		} 
		*
		[Calendar].[Year Month].[Year Month] *
		[Partner].[Partner Planning Code].[Partner Planning Code] *
		[Article].[Article Sku].[Article Sku]
		)
	ON ROWS	
FROM [Demand]
WHERE
	(
		[Fact Type].[Fact Type].&[2],
		[Calendar].[Year].&[2025]
	)
