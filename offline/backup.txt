//IBP выручка до конца года 1Q
CONCATENATE ([Revenue_and_Profit])
LOAD
    Date as Дата,
    sku as Артикул,
    'Выручка, руб без НДС' as "Коммерческий тип",
    Partner_Planning_Code as "Партнер планирования код",
    'Прогноз квартал' as "Сценарий",
    Value,
    Value as "Выручка 1Q до конца года, руб без НДС";
    
SELECT "Date",
    sku,
    "Partner_Planning_Code",
    "Code_CostItem",
    DemandScenarioName,
    "Value"
FROM "DWH_BI".bi."PNLIBP_v8"
INNER JOIN "DWH_BI".bi."vw_OperationExpenses_v3"
ON Code_CostItem = CodeExpensesPL
INNER JOIN "DWH_BI".bi."vw_demand_scenarios"
ON DemandScenarioName = Name
WHERE NameGroupExpenses1 = 'Выручка, руб без НДС'
--and NameCost='Выручка от реализации товара'
AND month("Date_start") = 1
AND Year("Date")= Year("Date_start")
AND Month("Date") IN (01,02,03,04,05,06,07,08,09,10,11,12)
AND "Partner_Planning_Code" not in ('ЕР-ЕР000387','ЕР-ЕР003862','ЕР-ЕР004646','ЦБ-ЦБ000357');
