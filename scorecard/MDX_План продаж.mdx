SELECT NON EMPTY
    {
	[Measures].[Plan Sales Quantity BUoM],
	[Measures].[Plan Sales Net Value]
	} ON COLUMNS,
      NON EMPTY(
			[Calendar Sales Report].[Year Month].[Year Month].members*
            [Partner].[Partner Planning Code].[Partner Planning Code]*
			--[Organization].[Organization].[Organization].members*
			[Article].[Article Sku].[Article Sku].members

            ) ON ROWS
FROM [Sales]
WHERE 
[Calendar Sales Report].[Year].[2024]
:
[Calendar Sales Report].[Year].[2025]
